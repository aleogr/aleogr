# The binding made right after the account is refused once (the new account is not
# visible yet): the run waits 30 seconds and tries again, and the second try works.
COUNT 2 ^iam service-accounts add-iam-policy-binding .*--role=roles/iam.workloadIdentityUser
COUNT 1 ^sleep 30$
ORDER ^iam service-accounts create demo-ci >> ^sleep 30$
CALLED ^iam service-accounts create demo-ci
NOT_CALLED ^sleep (60|1m)
OUTPUT "status": "created"
OUTPUT "condition": "as-declared"
OUTPUT_NOT NOT_FOUND
