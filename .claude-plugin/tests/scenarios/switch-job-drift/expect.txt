# The weeknight job runs at another time, and the weekend job lacks the method override
# header (it would send a plain POST and stop nothing): both are updated with the full set
# of values, nothing is created.
CALLED ^scheduler jobs update http aleogr-lab-stop-weeknight .*--schedule=30 22 \* \* 0-4
CALLED ^scheduler jobs update http aleogr-lab-stop-weeknight .*--uri=https://sqladmin.googleapis.com/v1/projects/aleogr-lab-shared-zz99/instances/aleogr-lab
CALLED ^scheduler jobs update http aleogr-lab-stop-weeknight .*--message-body-from-file=.*stop-body.json
CALLED ^scheduler jobs update http aleogr-lab-stop-weekend .*--http-method=POST
CALLED ^scheduler jobs update http aleogr-lab-stop-weekend .*--update-headers=[^ ]*X-HTTP-Method-Override=PATCH
NOT_CALLED ^scheduler jobs create
NOT_CALLED --http-method=(PATCH|PUT|patch|put)
OUTPUT "status": "updated"
OUTPUT "ready": true
