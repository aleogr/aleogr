# The instance is started; on the second try of the wait the describe fails (the shared project's
# policy cannot be read for a moment), on the third the database can be read: a failed read is a try
# like "not read yet" (the review's I2), so the run goes on and makes the database, the user, the
# roles and the settings, and the failed try's sentence is not in the answer.
ORDER ^scheduler jobs describe aleogr-shared-stop-weeknight --project=aleogr-shared-ab12 >> ^sql instances patch
ORDER ^sql instances patch aleogr-shared --project=aleogr-shared-ab12 --activation-policy=ALWAYS >> ^sleep 30$
ORDER ^sleep 30$ >> ^sql databases list
ORDER ^sql databases list >> ^sql databases create demo_dev
ORDER ^sql databases create >> ^projects add-iam-policy-binding aleogr-shared-ab12
ORDER ^projects add-iam-policy-binding >> ^run services update demo-web --update-env-vars=
AT_MOST 1 ^sql instances patch
COUNT 2 ^sleep 30$
COUNT 4 ^projects get-iam-policy aleogr-shared-ab12 --format=json
CALLED ^sql users create demo-web@aleogr-dev-demo-ab12\.iam

OUTPUT "status": "updated"
