# The first read is refused (role missing); the role is granted and the API enabled; the
# read right after is refused twice (the role has not propagated), the run waits 30
# seconds between tries and goes on when the read answers.
CALLED ^organizations add-iam-policy-binding 111111111111 .*--role=roles/iap\.settingsAdmin( |$)
CALLED ^services enable iap\.googleapis\.com
COUNT 4 ^iap settings get .*--billing-project=aleogr-shared-ab12( |$)
COUNT 2 ^sleep 30$
NOT_CALLED ^iap settings set
OUTPUT "status": "updated"
OUTPUT "prepared": false
OUTPUT "item": "login"
OUTPUT read -r -s -p
OUTPUT_NOT PERMISSION_DENIED
# Every wait is exactly sleep 30 (COUNT is "at least"; this keeps other lengths out).
NOT_CALLED ^sleep ([^3]|3[^0]|30.)
