# The role was just granted and has not propagated: the setting's read is refused twice,
# the run waits 30 seconds between tries and goes on when the read answers.
CALLED ^organizations add-iam-policy-binding 111111111111 .*--role=roles/iap\.settingsAdmin( |$)
ORDER ^organizations add-iam-policy-binding >> ^iap settings get
COUNT 3 ^iap settings get
COUNT 2 ^sleep 30$
NOT_CALLED ^iap settings set
OUTPUT "status": "updated"
OUTPUT "prepared": false
OUTPUT "item": "login"
OUTPUT read -s -p
OUTPUT_NOT PERMISSION_DENIED
